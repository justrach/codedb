    57	pub fn main(init: std.process.Init.Minimal) !void {
    58	    cio.setProcessArgs(cio.bootstrapArgs(init.args));
    59	    var gpa: std.heap.DebugAllocator(.{}) = .init;
    60	    defer _ = gpa.deinit();
   120	    }
   121	
   122	    var results: std.ArrayList(CaseResult) = .empty;
   123	    defer results.deinit(allocator);
   124	
   125	    for (cases, 0..) |case, idx| {
   126	        if (filter) |f| {
   127	            if (std.mem.indexOf(u8, case.name, f) == null) continue;
   128	        }
   129	        const args = &args_store[idx].value.object;
   130	        const r = try runCase(io, allocator, &bench_ctx, &store, &explorer, &agents, case, args, &telem_off, huge_content);
   131	        try results.append(allocator, .{
   132	            .tool = case.name,
   133	            .avg_latency_ns = r.avg_latency_ns,
   134	            .min_latency_ns = r.min_latency_ns,
   135	            .response_bytes = r.response_bytes,
   136	            .ops_per_sec = opsPerSec(r.avg_latency_ns),
   137	        });
   138	    }
   139	
   140	    const corpus = summarizeCorpus(&explorer);
   141	    try writeHumanSummary(allocator, cio.File.stderr(), corpus.files, corpus.bytes, scan_ns, results.items);
   142	    if (emit_json) {
   143	        try writeJsonSummary(allocator, cio.File.stdout(), repo_root, tmp_root, corpus.files, corpus.bytes, scan_ns, results.items);
   144	    }
   145	}

   149	fn bustSearchCache(explorer: *Explorer) !void {
   150	    bust_flip = !bust_flip;
   151	    const content = if (bust_flip) "pub const bust_marker = 1;\n" else "pub const bust_marker = 2;\n";
   152	    try explorer.indexFile("bustfile.zig", content);

   155	fn resetHuge(explorer: *Explorer, store: *Store, huge_content: []const u8) !void {
   156	    try explorer.indexFile("huge.zig", huge_content);
   157	    _ = try store.recordSnapshot("huge.zig", huge_content.len, std.hash.Wyhash.hash(0, huge_content));
   158	}

   160	fn runCase(
   161	    io: std.Io,
   162	    allocator: std.mem.Allocator,
   163	    bench_ctx: *mcp.BenchContext,
   164	    store: *Store,
   165	    explorer: *Explorer,
   166	    agents: *AgentRegistry,
   167	    case: Case,
   168	    args: *const std.json.ObjectMap,
   169	    telem: *telemetry.Telemetry,
   170	    huge_content: []const u8,
   171	) !struct { avg_latency_ns: u64, min_latency_ns: u64, response_bytes: usize } {
   172	    var total_ns: u64 = 0;
   173	    var min_ns: u64 = std.math.maxInt(u64);
   174	    var response_bytes: usize = 0;
   175	
   176	    for (0..case.warmup) |_| {
   177	        if (case.bust_search_cache) try bustSearchCache(explorer);
   178	        if (case.reset_huge) try resetHuge(explorer, store, huge_content);
   179	        _ = bench_ctx.runToolCall(io, allocator, case.name, case.tool, args, store, explorer, agents, telem);
   180	    }
   181	
   182	    if (cio.posixGetenv("CODEDB_EDGE_DUMP") != null) {
   183	        var dump_out: std.ArrayList(u8) = .empty;
   184	        defer dump_out.deinit(allocator);
   185	        bench_ctx.runDispatch(io, allocator, case.tool, args, &dump_out, store, explorer, agents);
   186	        cio.File.stderr().writeAll("=== DUMP ") catch {};
   187	        cio.File.stderr().writeAll(case.name) catch {};
   188	        cio.File.stderr().writeAll(" ===\n") catch {};
   189	        cio.File.stderr().writeAll(dump_out.items) catch {};
   190	        cio.File.stderr().writeAll("\n=== END ===\n") catch {};
   191	    }
   192	
   193	    for (0..case.iterations) |_| {
   194	        if (case.bust_search_cache) try bustSearchCache(explorer);
   195	        if (case.reset_huge) try resetHuge(explorer, store, huge_content);
   196	
   197	        const r = bench_ctx.runToolCall(io, allocator, case.name, case.tool, args, store, explorer, agents, telem);
   198	        total_ns +|= r.dispatch_ns;
   199	        if (r.dispatch_ns < min_ns) min_ns = r.dispatch_ns;
   200	        response_bytes = r.response_bytes;
   201	    }
   202	
   203	    return .{
   204	        .avg_latency_ns = @intCast(@divTrunc(total_ns, case.iterations)),
   205	        .min_latency_ns = min_ns,
   206	        .response_bytes = response_bytes,
   207	    };
   208	}

   210	fn genCorpus(io: std.Io, allocator: std.mem.Allocator, tmp_root: []const u8) ![]const u8 {
   211	    var buf: std.ArrayList(u8) = .empty;
   212	    defer buf.deinit(allocator);
   213	    const w = cio.listWriter(&buf, allocator);
   214	
   215	    // huge.zig: ~1.8MB, ~47k lines, 3600 functions. Above the 1MB trigram cap,
   216	    // below the 2MB index cap — lands in the skip-trigram full-scan set.
   217	    try buf.ensureTotalCapacity(allocator, 2_000_000);
   218	    for (0..HUGE_FN_COUNT) |i| {
   219	        try w.print("pub fn edgeHugeFn_{d:0>4}(x: usize) usize {{\n", .{i});
   220	        try w.writeAll("    var acc: usize = x;\n");
