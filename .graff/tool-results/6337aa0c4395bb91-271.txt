  438 /// Pure decision function — useful for tests. Returns true when the next
  439 /// auto-update attempt should fire.
  440 pub fn shouldRunAutoUpdate(now_ms: i64, last_check_ms: ?i64, env_disabled: bool) bool {
  441     if (env_disabled) return false;
  442     const last = last_check_ms orelse return true;
  443     if (last > now_ms) return true;
  444     const delta: i128 = @as(i128, now_ms) - @as(i128, last);
  445     return delta >= auto_update_throttle_ms;
  446 }

  449     const env_disabled = cio.posixGetenv("CODEDB_NO_AUTO_UPDATE") != null;
  450     const home = cio.homeDir() orelse return;

  454     defer allocator.free(dir_path);

  456 
  457     const stamp_path = std.fmt.allocPrint(allocator, "{s}/{s}", .{ dir_path, auto_update_stamp_filename }) catch return;
  458     defer allocator.free(stamp_path);
  459 
  460     const last_ms = readAutoUpdateStamp(io, stamp_path);
  461     const now_ms = cio.milliTimestamp();

  464     const executable_path = std.process.executablePathAlloc(io, std.heap.page_allocator) catch return;
  465     const thread = std.Thread.spawn(.{}, autoUpdateWorker, .{executable_path}) catch {
  466         std.heap.page_allocator.free(executable_path);

  468     };
  469     thread.detach();
  470 
  471     // Only throttle after the worker launches so a local spawn failure can be
  472     // retried on the next MCP startup.
  473     writeAutoUpdateStamp(io, stamp_path, now_ms);
