  249 | pub fn runCliDaemon(ctx: *RunCtx) !void {
  250 |     const io = ctx.io;
  251 |     const allocator = ctx.allocator;
  252 |     const abs_root = ctx.abs_root;
  253 |     const data_dir = ctx.data_dir;
  254 |     const root = ctx.root;
  255 |     const store = ctx.store;
  256 |     const explorer = ctx.explorer;
  257 |     const out = ctx.out;
  258 |     // Hidden command: a lightweight warm daemon spawned by a cold CLI query
  259 |     // so the NEXT query is fast. It is `serve` minus the TCP server, agent
  260 |     // registry, and reaper — just the warm explorer/store (loaded by the
  261 |     // section above), an incremental watcher to keep the index fresh, the
  262 |     // per-project CLI socket listener, and a time-based idle watchdog that
  263 |     // exits the process once the socket has been quiet for a while. We do
  264 |     // NOT auto-spawn `serve` here because two `serve` daemons would fight
  265 |     // over the fixed TCP port; the CLI socket is per-project and conflict-free.
  266 | 
  267 |     // Detach from the controlling terminal so we outlive the spawning CLI
  268 |     // and never touch its stdio. (spawnDetached already pointed 0/1/2 at
  269 |     // /dev/null; this also starts a fresh session.)
  270 |     cio.detachFromTerminal();
  271 | 
  272 |     const idle_ms: i64 = blk: {
  273 |         const raw = cio.posixGetenv("CODEDB_CLI_DAEMON_IDLE_MS") orelse break :blk 5 * 60 * 1000;
  274 |         break :blk std.fmt.parseInt(i64, raw, 10) catch (5 * 60 * 1000);
  275 |     };
  276 | 
  277 |     var shutdown = std.atomic.Value(bool).init(false);
  278 |     defer shutdown.store(true, .release);
  279 |     // The load section above already loaded/scanned the index, so the
  280 |     // watcher starts in the "scan done" state and only does incremental
  281 |     // upkeep from here.
  282 |     var scan_already_done = std.atomic.Value(bool).init(true);
  283 | 
  284 |     // Grace period: treat startup as activity so a freshly-spawned daemon
  285 |     // gets the full idle window before the watchdog can fire.
  286 |     var last_activity_ms = std.atomic.Value(i64).init(cio.milliTimestamp());
  287 | 
  288 |     const queue = try allocator.create(watcher.EventQueue);
  289 |     defer allocator.destroy(queue);
  290 |     queue.* = watcher.EventQueue{};
  291 |     const watch_thread = try std.Thread.spawn(.{}, watcher.incrementalLoop, .{ io, store, explorer, queue, root, &shutdown, &scan_already_done });
  292 |     watch_thread.detach();
  293 | 
  294 |     std.log.info("cli-daemon: {d} files indexed, idle_timeout={d}ms", .{ store.currentSeq(), idle_ms });
  295 | 
  296 |     // CLI socket listener. Pass the REAL shutdown flag: if another daemon
  297 |     // already owns the socket (we lost the bind race), cliDaemonListen sets
  298 |     // shutdown and the watchdog below returns at once, so the redundant
  299 |     // daemon exits instead of lingering idle.
  300 |     if (std.Thread.spawn(.{}, cliDaemonListen, .{ io, allocator, explorer, store, abs_root, data_dir, &last_activity_ms, &shutdown, false })) |cli_t| {
  301 |         cli_t.detach();
  302 |     } else |err| {
  303 |         std.log.warn("cli-proxy: could not start listener: {s}", .{@errorName(err)});
  304 |         return;
  305 |     }
  306 | 
  307 |     // Warm the word index + result caches in the background so the first
  308 |     // proxied CLI queries hit a fully warm explorer (the daemon used to
  309 |     // stay lean and charge the first `word`/`context` query the rebuild).
  310 |     spawnWarmup(io, allocator, explorer, data_dir, abs_root, &shutdown);
  311 | 
  312 |     // Block here until idle (or a bind-race shutdown). On return the process
  313 |     // exits immediately — std.process.exit reclaims everything and avoids
  314 |     // racing the detached listener thread against freed explorer/store.
  315 |     const idle_exit = cliIdleWatchdog(&shutdown, &last_activity_ms, idle_ms);
  316 |     shutdown.store(true, .release);
  317 |     // If WE owned the socket (exited on idle, not a bind-race loss), unlink
  318 |     // it on the way out. The listener thread's own `defer unlink` never runs
  319 |     // because std.process.exit kills it mid-accept; do it here so we don't
  320 |     // leave a stale node behind. On a bind-race loss the socket belongs to
  321 |     // the winning daemon, so idle_exit is false and we leave it alone.
  322 |     if (idle_exit) {
  323 |         var sock_buf: [128]u8 = undefined;
  324 |         if (cliSocketPath(&sock_buf, abs_root)) |sock_path| {
  325 |             var sock_z_buf: [128]u8 = undefined;
  326 |             if (cio.bufPrintZ(&sock_z_buf, "{s}", .{sock_path})) |sock_z| {
  327 |                 _ = std.c.unlink(sock_z.ptr);
  328 |             } else |_| {}
  329 |         }
  330 |     }
  331 |     out.flush();
  332 |     std.process.exit(0);
  333 | }
