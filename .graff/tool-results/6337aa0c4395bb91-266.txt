  105 |     fn save(name: []const u8) EnvVarGuard {
  106 |         var g = EnvVarGuard{ .name = name, .had_prev = false, .prev_len = 0, .prev = undefined };
  107 |         if (cio.posixGetenv(name)) |v| {
  108 |             if (v.len <= g.prev.len) {
  109 |                 @memcpy(g.prev[0..v.len], v);
  110 |                 g.prev_len = v.len;
  111 |                 g.had_prev = true;
  112 |             }
  113 |         }
  114 |         return g;
  115 |     }

  117 |     fn restore(self: *const EnvVarGuard) void {
  118 |         if (self.had_prev) {
  119 |             cio.posixSetenv(self.name, self.prev[0..self.prev_len]);
  120 |         } else {
  121 |             cio.posixUnsetenv(self.name);
  122 |         }
  123 |     }
