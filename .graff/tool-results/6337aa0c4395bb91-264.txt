  196 |     var target_it = std.mem.splitScalar(u8, trimVersionPrefix(target), '.');

  423 | // ── Auto-update ──────────────────────────────────────────────────────────────
  424 | //
  425 | // On startup we kick off a detached invocation of the built-in, checksum-
  426 | // verified updater. The check is:
  427 | //   • disabled when CODEDB_NO_AUTO_UPDATE is set in the environment
  428 | //   • throttled to at most once every 24 hours, tracked in
  429 | //     ~/.codedb/last_auto_update_check (a u64 little-endian unix-ms timestamp)
  430 | // The update itself runs in a background thread so network failures never block
  431 | // server startup. If the binary on disk gets replaced mid-session, the kernel
  432 | // keeps the old inode mapped for the running process; subsequent invocations
  433 | // get the new binary.
  434 | 
  435 | const auto_update_throttle_ms: i64 = 24 * 60 * 60 * 1000;
  436 | const auto_update_stamp_filename = "last_auto_update_check";
  437 | 
  438 | /// Pure decision function — useful for tests. Returns true when the next
  439 | /// auto-update attempt should fire.
  440 | pub fn shouldRunAutoUpdate(now_ms: i64, last_check_ms: ?i64, env_disabled: bool) bool {
  441 |     if (env_disabled) return false;
  442 |     const last = last_check_ms orelse return true;
  443 |     if (last > now_ms) return true;
  444 |     const delta: i128 = @as(i128, now_ms) - @as(i128, last);
  445 |     return delta >= auto_update_throttle_ms;
  446 | }

  448 | pub fn maybeAutoUpdate(io: std.Io, allocator: std.mem.Allocator) void {
  449 |     const env_disabled = cio.posixGetenv("CODEDB_NO_AUTO_UPDATE") != null;

  451 |     if (home.len == 0) return;

  454 |     defer allocator.free(dir_path);
  455 |     std.Io.Dir.cwd().createDirPath(io, dir_path) catch {};

  458 |     defer allocator.free(stamp_path);

  466 |         std.heap.page_allocator.free(executable_path);
  467 |         return;
  468 |     };
  469 |     thread.detach();

  471 |     // Only throttle after the worker launches so a local spawn failure can be
  472 |     // retried on the next MCP startup.
