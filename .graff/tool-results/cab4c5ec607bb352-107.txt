    57	pub fn main(init: std.process.Init.Minimal) !void {
    58	    cio.setProcessArgs(cio.bootstrapArgs(init.args));
    59	    var gpa: std.heap.DebugAllocator(.{}) = .init;
    60	    defer _ = gpa.deinit();
    61	    const allocator = gpa.allocator();
    62	
    63	    var threaded: std.Io.Threaded = .init(allocator, .{});
    64	    defer threaded.deinit();
    65	    const io = threaded.io();
    66	
    67	    var emit_json = false;
    68	    var filter: ?[]const u8 = null;
    69	    {
    70	        const args = try cio.argsAlloc(allocator);
    71	        defer cio.argsFree(allocator, args);
    72	        for (args[1..]) |arg| {
    73	            if (std.mem.eql(u8, arg, "--json")) emit_json = true;
    74	            if (std.mem.startsWith(u8, arg, "--filter=")) {
    75	                filter = try allocator.dupe(u8, arg["--filter=".len..]);
    76	            }
    77	        }
    78	    }
    79	    defer if (filter) |f| allocator.free(f);
    80	
    81	    var tmp_path_buf: [std.fs.max_path_bytes]u8 = undefined;
    82	    const tmp_root = try makeTempCorpusDir(io, &tmp_path_buf);
    83	    defer std.Io.Dir.cwd().deleteTree(io, tmp_root) catch {};
    84	
    85	    var repo_path_buf: [std.fs.max_path_bytes]u8 = undefined;
    86	    const repo_root_len = try std.Io.Dir.cwd().realPathFile(io, ".", &repo_path_buf);
    87	    const repo_root = repo_path_buf[0..repo_root_len];
    88	
    89	    const huge_content = try genCorpus(io, allocator, tmp_root);
    90	    defer allocator.free(huge_content);
    91	
    92	    var store = Store.init(allocator);
    93	    defer store.deinit();
    94	
    95	    var explorer = Explorer.init(allocator, Explorer.DEFAULT_CONTENT_CACHE_CAPACITY);
    96	    defer explorer.deinit();
    97	
    98	    var agents = AgentRegistry.init(allocator);
    99	    defer agents.deinit();
   100	    _ = try agents.register("__filesystem__");
   101	
   102	    const scan_t0 = cio.nanoTimestamp();
   103	    try watcher.initialScan(io, &store, &explorer, tmp_root, allocator, false);
   104	    const scan_ns: u64 = @intCast(cio.nanoTimestamp() - scan_t0);
   105	
   106	    var bench_ctx = mcp.BenchContext.init(allocator, tmp_root, Explorer.DEFAULT_CONTENT_CACHE_CAPACITY);
   107	    defer bench_ctx.deinit();
   108	
   109	    try std.process.setCurrentPath(io, tmp_root);
   110	    defer std.process.setCurrentPath(io, repo_root) catch {};
