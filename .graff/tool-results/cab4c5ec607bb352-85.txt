    70	/// Classify `path` for the rerank path priors. Pure — the result is cached
    71	/// on FileOutline at init; callers without an outline compute it directly.
    72	pub fn classifyPath(path: []const u8) PathClass {
    73	    var c: PathClass = .{};
    74	    // One tokenize pass over the path instead of one per segment keyword.
    75	    var iter = std.mem.tokenizeAny(u8, path, "/\\");
    76	    while (iter.next()) |seg| {
    77	        // #598: mention-dense tooling files (a bench script repeating the
    78	        // term six times per line) saturate the per-line count and shrug off
    79	        // the ×0.5 path prior — the occurrence BASE is capped for tooling.
    80	        if (std.mem.eql(u8, seg, "bench") or std.mem.eql(u8, seg, "benchmarks") or
   180	    php,
   181	    ruby,
   182	    hcl,
   183	    r,
   184	    markdown,
   185	    json,
   186	    yaml,
   187	    unknown,
   188	    dart,
   189	    java,
   190	    kotlin,
   191	    swift,
   192	    svelte,
   193	    vue,
   194	    astro,
   195	    shell,
   196	    css,
   197	    scss,
   198	    sql,
   199	    protobuf,
   200	    fortran,
   201	    llvm_ir,
   202	    mlir,
   203	    tablegen,
   204	    rescript,
   205	    ocaml,
   206	};

   208	pub fn detectLanguage(path: []const u8) Language {
   209	    if (std.mem.endsWith(u8, path, ".zig")) return .zig;
   210	    if (std.mem.endsWith(u8, path, ".c") or std.mem.endsWith(u8, path, ".h")) return .c;
   211	    if (std.mem.endsWith(u8, path, ".cpp") or std.mem.endsWith(u8, path, ".hpp") or
   212	        std.mem.endsWith(u8, path, ".cc") or std.mem.endsWith(u8, path, ".hh") or
   213	        std.mem.endsWith(u8, path, ".cxx") or std.mem.endsWith(u8, path, ".hxx") or
   214	        std.mem.endsWith(u8, path, ".mm"))
   215	        return .cpp;
   216	    if (std.mem.endsWith(u8, path, ".py")) return .python;
   217	    if (std.mem.endsWith(u8, path, ".js") or std.mem.endsWith(u8, path, ".jsx")) return .javascript;
   218	    if (std.mem.endsWith(u8, path, ".ts") or std.mem.endsWith(u8, path, ".tsx")) return .typescript;
   219	    if (std.mem.endsWith(u8, path, ".rs")) return .rust;
   220	    if (std.mem.endsWith(u8, path, ".go")) return .go_lang;
   221	    if (std.mem.endsWith(u8, path, ".php")) return .php;
   222	    if (std.mem.endsWith(u8, path, ".rb") or std.mem.endsWith(u8, path, ".rake")) return .ruby;
   223	    if (std.mem.endsWith(u8, path, ".tf") or std.mem.endsWith(u8, path, ".tfvars") or std.mem.endsWith(u8, path, ".hcl")) return .hcl;
   224	    if (std.mem.endsWith(u8, path, ".r") or std.mem.endsWith(u8, path, ".R")) return .r;
   225	    if (std.mem.endsWith(u8, path, ".md")) return .markdown;
   226	    if (std.mem.endsWith(u8, path, ".json")) return .json;
   227	    if (std.mem.endsWith(u8, path, ".yaml") or std.mem.endsWith(u8, path, ".yml")) return .yaml;
   228	    if (std.mem.endsWith(u8, path, ".dart")) return .dart;
   229	    if (std.mem.endsWith(u8, path, ".java")) return .java;
   230	    if (std.mem.endsWith(u8, path, ".kt")) return .kotlin;
   231	    if (std.mem.endsWith(u8, path, ".swift")) return .swift;
   232	    if (std.mem.endsWith(u8, path, ".svelte")) return .svelte;
   233	    if (std.mem.endsWith(u8, path, ".vue")) return .vue;
   234	    if (std.mem.endsWith(u8, path, ".astro")) return .astro;
   235	    if (std.mem.endsWith(u8, path, ".sh")) return .shell;
   236	    if (std.mem.endsWith(u8, path, ".css")) return .css;
   237	    if (std.mem.endsWith(u8, path, ".scss")) return .scss;
   238	    if (std.mem.endsWith(u8, path, ".sql")) return .sql;
   239	    if (std.mem.endsWith(u8, path, ".proto")) return .protobuf;
   240	    if (std.mem.endsWith(u8, path, ".f90")) return .fortran;
   241	    if (std.mem.endsWith(u8, path, ".ll")) return .llvm_ir;
   242	    if (std.mem.endsWith(u8, path, ".mlir")) return .mlir;
   243	    if (std.mem.endsWith(u8, path, ".td")) return .tablegen;
   244	    if (std.mem.endsWith(u8, path, ".res") or std.mem.endsWith(u8, path, ".resi")) return .rescript;
   245	    if (std.mem.endsWith(u8, path, ".ml") or std.mem.endsWith(u8, path, ".mli")) return .ocaml;
   246	    return .unknown;
   247	}

   249	/// Returns true for languages whose content is primarily prose / data /
   250	/// markup rather than executable code. Used to deprioritise these files in
   251	/// content search so a CHANGELOG.md or design doc cannot starve a canonical
   252	/// source-file match (issue #430).
   253	pub fn isDocLanguage(lang: Language) bool {
   254	    return switch (lang) {
   255	        .markdown, .json, .yaml, .unknown => true,
   256	        else => false,
   257	    };
   258	}
