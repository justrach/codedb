  140 | pub fn compareVersions(current: []const u8, target: []const u8) !std.math.Order {
  141 |     var current_it = std.mem.splitScalar(u8, trimVersionPrefix(current), '.');
  142 |     var target_it = std.mem.splitScalar(u8, trimVersionPrefix(target), '.');
  143 | 
  144 |     while (true) {
  145 |         const current_part = current_it.next();
  146 |         const target_part = target_it.next();
  147 | 
  148 |         if (current_part == null and target_part == null) return .eq;
  149 | 
  150 |         const current_num = if (current_part) |part| try parseVersionPart(part) else 0;
  151 |         const target_num = if (target_part) |part| try parseVersionPart(part) else 0;
  152 | 
  153 |         if (current_num < target_num) return .lt;
  154 |         if (current_num > target_num) return .gt;
  155 |     }

  354 | // ── Auto-update ──────────────────────────────────────────────────────────────
  355 | //
  356 | // On startup we kick off a detached invocation of the built-in, checksum-
  357 | // verified updater. The check is:
  358 | //   • disabled when CODEDB_NO_AUTO_UPDATE is set in the environment
  359 | //   • throttled to at most once every 24 hours, tracked in
  360 | //     ~/.codedb/last_auto_update_check (a u64 little-endian unix-ms timestamp)
  361 | // The update itself runs in a background thread so network failures never block
  362 | // server startup. If the binary on disk gets replaced mid-session, the kernel
  363 | // keeps the old inode mapped for the running process; subsequent invocations
  364 | // get the new binary.
  365 | 
  366 | const auto_update_throttle_ms: i64 = 24 * 60 * 60 * 1000;
  367 | const auto_update_stamp_filename = "last_auto_update_check";
  368 | 
  369 | /// Pure decision function — useful for tests. Returns true when the next
  370 | /// auto-update attempt should fire.
  371 | pub fn shouldRunAutoUpdate(now_ms: i64, last_check_ms: ?i64, env_disabled: bool) bool {
  372 |     if (env_disabled) return false;
  373 |     const last = last_check_ms orelse return true;
  374 |     if (last > now_ms) return true;
  375 |     const delta: i128 = @as(i128, now_ms) - @as(i128, last);
  376 |     return delta >= auto_update_throttle_ms;
  377 | }

  379 | pub fn maybeAutoUpdate(io: std.Io, allocator: std.mem.Allocator) void {
  380 |     const env_disabled = cio.posixGetenv("CODEDB_NO_AUTO_UPDATE") != null;
  381 |     const home = cio.homeDir() orelse return;
  382 |     if (home.len == 0) return;
  383 | 
  384 |     const dir_path = std.fmt.allocPrint(allocator, "{s}/.codedb", .{home}) catch return;
  385 |     defer allocator.free(dir_path);
  386 |     std.Io.Dir.cwd().createDirPath(io, dir_path) catch {};
  387 | 
  388 |     const stamp_path = std.fmt.allocPrint(allocator, "{s}/{s}", .{ dir_path, auto_update_stamp_filename }) catch return;
  389 |     defer allocator.free(stamp_path);
  390 | 
  391 |     const last_ms = readAutoUpdateStamp(io, stamp_path);
  392 |     const now_ms = cio.milliTimestamp();
  393 |     if (!shouldRunAutoUpdate(now_ms, last_ms, env_disabled)) return;
  394 | 
  395 |     const executable_path = std.process.executablePathAlloc(io, std.heap.page_allocator) catch return;
  396 |     const thread = std.Thread.spawn(.{}, autoUpdateWorker, .{executable_path}) catch {
  397 |         std.heap.page_allocator.free(executable_path);
  398 |         return;
  399 |     };
  400 |     thread.detach();
  401 | 
  402 |     // Only throttle after the worker launches so a local spawn failure can be
  403 |     // retried on the next MCP startup.
  404 |     writeAutoUpdateStamp(io, stamp_path, now_ms);
  405 | }
