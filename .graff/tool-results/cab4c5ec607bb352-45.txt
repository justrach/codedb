  2560	    }
  2561	
  2562	    // #573: single filter pass — the header count and the printed entries come
  2563	    // from the same accumulation, so a predicate edit cannot desync them (the
  2564	    // previous count loop + print loop duplicated four predicates verbatim).
  2565	    var kept: std.ArrayList(usize) = .empty;
  2566	    defer kept.deinit(alloc);
  2567	    for (results, 0..) |r, r_idx| {
  2568	        if (kept.items.len >= max_results) break;
  2569	        const lang = explore_mod.detectLanguage(r.path);
  2570	        if (!langHasCallSites(lang)) continue;
  2571	        // #562: a full-line comment mention is documentation, not a call site.
  2572	        if (explore_mod.isCommentOrBlank(r.line_text, lang)) continue;
  2573	        var is_def = false;
  2574	        for (defs) |d| {
  2575	            if (r.line_num == d.symbol.line_start and std.mem.eql(u8, r.path, d.path)) {
  2576	                is_def = true;
  2577	                break;
  2578	            }
  2579	        }
  2580	        if (is_def) continue;
  2581	        // A whole-word match that only ever lands inside a string literal or a
  2582	        // trailing comment is a mention, not an invocation — a `("name",
  2583	        // "file")` table row, a `test "…name…" {` declaration, or
  2584	        // `init(); // then name() …`. Real calls sit on the code prefix of the
  2585	        // line, outside the quotes, and survive both cuts. (#682)
  2586	        if (!hasWholeWordMatchOutsideStrings(lineCodePrefix(r.line_text, lang), name)) continue;
  2587	        kept.append(alloc, r_idx) catch {};
  2588	    }
  2589	
  2590	    const w = cio.listWriter(out, alloc);
  2591	    w.print("{d} call sites for '{s}':\n", .{ kept.items.len, name }) catch {};
  2592	    for (kept.items) |kept_idx| {
  2593	        const r = results[kept_idx];
  2594	        if (r.scope_name) |sn| {
  2595	            appendMatchLineNoNL(w, "  ", r.path, r.line_num, r.line_text);
  2596	            w.print("  [in {s} ({s}, L{d}-L{d})]\n", .{ sn, @tagName(r.scope_kind.?), r.scope_start, r.scope_end }) catch {};
  2597	        } else {
  2598	            appendMatchLine(w, "  ", r.path, r.line_num, r.line_text);
  2599	        }
  2600	    }
  2601	}

  2603	fn handleCallpath(alloc: std.mem.Allocator, args: *const std.json.ObjectMap, out: *std.ArrayList(u8), explorer: *Explorer) void {
  2604	    const from_name = getStr(args, "from") orelse {
  2605	        out.appendSlice(alloc, "error: missing 'from' argument") catch {};
  2606	        appendBundleArgKeysDiagnostic(alloc, out, args);
  2607	        return;
  2608	    };
  2609	    const to_name = getStr(args, "to") orelse {
  2610	        out.appendSlice(alloc, "error: missing 'to' argument") catch {};
  2611	        appendBundleArgKeysDiagnostic(alloc, out, args);
  2612	        return;
  2613	    };
  2614	    if (from_name.len == 0 or to_name.len == 0) {
  2615	        out.appendSlice(alloc, "error: 'from' and 'to' must be non-empty symbol names") catch {};
  2616	        return;
  2617	    }
  2618	    const max_hops: usize = if (getInt(args, "max_hops")) |n| @intCast(@max(1, @min(n, 64))) else 12;
  2619	
  2620	    const steps = explorer.findCallPath(from_name, to_name, alloc, max_hops) catch {
  2621	        out.appendSlice(alloc, "error: callpath search failed") catch {};
  2622	        return;
  2623	    };
  2624	    const path = steps orelse {
  2625	        const w = cio.listWriter(out, alloc);
  2626	        w.print("no call path from '{s}' to '{s}' within {d} hops\n", .{ from_name, to_name, max_hops }) catch {};
  2627	        return;
  2628	    };
  2629	    defer alloc.free(path);
  2630	
  2631	    const w = cio.listWriter(out, alloc);
  2632	    w.print("call path ({d} hops): {s} → {s}\n", .{ path.len - 1, from_name, to_name }) catch {};
  2633	    for (path, 0..) |step, i| {
  2634	        if (i > 0) w.print("  → ", .{}) catch {};
  2635	        w.print("{s}:{s}@L{d}", .{ step.path, step.name, step.line }) catch {};
  2636	        if (i + 1 < path.len) w.print("\n", .{}) catch {};
  2637	    }
  2638	    w.print("\n", .{}) catch {};
  2639	}

  2642	    return (c >= 'a' and c <= 'z') or

  2648	/// Returns true iff `needle` occurs in `line` as a whole-word identifier
  2649	/// (non-identifier characters or line boundaries on both sides) at a position
  2650	/// that is NOT inside a string literal.
  2651	///
  2652	/// An occurrence that only ever appears between quotes is a mention, not a
  2653	/// call: the Python table row `("renderPlainSearch", "src/explore.zig"),` and
  2654	/// the Zig declaration `test "def-first: renderPlainSearch surfaces …" {` both
  2655	/// name the symbol without invoking it, yet passed every other filter here.
  2656	///
  2657	/// Deliberately line-local and pragmatic, in the spirit of `isCommentOrBlank`:
  2658	///   - tracks '…' and "…" spans, honouring backslash escapes
  2659	///   - a quote with no closing partner on the line (an apostrophe in a
  2660	///     trailing comment, a Rust lifetime `&'a`, an open multi-line literal) is
  2661	///     treated as ordinary code, so it can never swallow the rest of the line
  2662	///   - backticks are not treated as quotes: Go raw strings and JS template
  2663	///     literals routinely wrap real calls (`${render()}`)
  2664	fn hasWholeWordMatchOutsideStrings(line: []const u8, needle: []const u8) bool {
  2665	    if (needle.len == 0 or line.len < needle.len) return false;
  2666	    var i: usize = 0;
  2667	    while (i < line.len) {
  2668	        if (line[i] == '"' or line[i] == '\'') {
  2669	            if (stringLiteralEnd(line, i)) |end| {
  2670	                i = end + 1;
  2671	                continue;
  2672	            }
  2673	        }
  2674	        // `i` is a code byte — try to anchor a whole-word match on it.
  2675	        if (line.len - i >= needle.len and std.mem.eql(u8, line[i..][0..needle.len], needle)) {
  2676	            const before_ok = i == 0 or !isIdentChar(line[i - 1]);
  2677	            const after = i + needle.len;
  2678	            const after_ok = after >= line.len or !isIdentChar(line[after]);
  2679	            if (before_ok and after_ok) return true;
  2680	        }
  2681	        i += 1;
  2682	    }
  2683	    return false;
  2684	}

  2686	/// The slice of `line` before its first line-comment marker outside string
  2687	/// literals, per `language` — the part of the line that can actually contain a
  2688	/// call. Lines with no marker, and languages without a line-comment syntax
  2689	/// (OCaml's `(* *)` is block-only), come back whole. (#682)
  2690	///
  2691	/// Same line-local scanning discipline as `hasWholeWordMatchOutsideStrings`:
  2692	/// quoted spans are skipped (so `"https://…"` never truncates), an unclosed
  2693	/// quote is ordinary code. Mid-line block-comment openers are left alone — a
  2694	/// false negative re-admits a mention; truncating at `/*` could drop the real
  2695	/// call in `foo(); /* … */ bar()`.
  2696	fn lineCodePrefix(line: []const u8, language: explore_mod.Language) []const u8 {
  2697	    const markers: []const []const u8 = switch (language) {
  2698	        .zig, .rust, .go_lang, .javascript, .typescript, .c, .cpp, .dart, .java, .kotlin, .mlir, .tablegen, .rescript, .svelte, .vue, .astro => &.{"//"},
  2699	        .python, .ruby, .r, .shell => &.{"#"},
  2700	        .hcl => &.{ "#", "//" },
  2701	        .sql => &.{"--"},
  2702	        .fortran => &.{"!"},
  2703	        .llvm_ir => &.{";"},
  2704	        else => return line,
  2705	    };
  2706	    var i: usize = 0;
  2707	    while (i < line.len) {
  2708	        if (line[i] == '"' or line[i] == '\'') {
  2709	            if (stringLiteralEnd(line, i)) |end| {
  2710	                i = end + 1;
  2711	                continue;
  2712	            }
  2713	        }
  2714	        for (markers) |m| {
  2715	            if (std.mem.startsWith(u8, line[i..], m)) {
  2716	                // Shell: `#` only opens a comment at the start of a word —
  2717	                // `${#var}` and `$#` are live syntax, not comments.
  2718	                if (language == .shell and i != 0 and line[i - 1] != ' ' and line[i - 1] != '\t') break;
  2719	                return line[0..i];
  2720	            }
  2721	        }
  2722	        i += 1;
  2723	    }
  2724	    return line;
  2725	}

  2727	/// Index of the quote closing the literal opened at `open`, or null when the
  2728	/// literal never closes on this line.
  2729	fn stringLiteralEnd(line: []const u8, open: usize) ?usize {
  2730	    const quote = line[open];
  2731	    var i = open + 1;
  2732	    while (i < line.len) : (i += 1) {
  2733	        if (line[i] == '\\') {
  2734	            i += 1;
  2735	            continue;
  2736	        }
  2737	        if (line[i] == quote) return i;
  2738	    }
  2739	    return null;
  2740	}
