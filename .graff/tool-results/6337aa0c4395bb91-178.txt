353:// ── Auto-update ──────────────────────────────────────────────────────────────
354://
355:// On startup we kick off a detached invocation of the built-in, checksum-
356:// verified updater. The check is:
357://   • disabled when CODEDB_NO_AUTO_UPDATE is set in the environment
358://   • throttled to at most once every 24 hours, tracked in
359://     ~/.codedb/last_auto_update_check (a u64 little-endian unix-ms timestamp)
360:// The update itself runs in a background thread so network failures never block
361:// server startup. If the binary on disk gets replaced mid-session, the kernel
362:// keeps the old inode mapped for the running process; subsequent invocations
363:// get the new binary.
364:
365:const auto_update_throttle_ms: i64 = 24 * 60 * 60 * 1000;
366:const auto_update_stamp_filename = "last_auto_update_check";
367:
368:/// Pure decision function — useful for tests. Returns true when the next
369:/// auto-update attempt should fire.
370:pub fn shouldRunAutoUpdate(now_ms: i64, last_check_ms: ?i64, env_disabled: bool) bool {
371:    if (env_disabled) return false;
372:    const last = last_check_ms orelse return true;
373:    if (last > now_ms) return true;
374:    const delta: i128 = @as(i128, now_ms) - @as(i128, last);
375:    return delta >= auto_update_throttle_ms;
376:}

378:pub fn maybeAutoUpdate(io: std.Io, allocator: std.mem.Allocator) void {
379:    const env_disabled = cio.posixGetenv("CODEDB_NO_AUTO_UPDATE") != null;
380:    const home = cio.homeDir() orelse return;
381:    if (home.len == 0) return;
382:
383:    const dir_path = std.fmt.allocPrint(allocator, "{s}/.codedb", .{home}) catch return;
384:    defer allocator.free(dir_path);
385:    std.Io.Dir.cwd().createDirPath(io, dir_path) catch {};
386:
387:    const stamp_path = std.fmt.allocPrint(allocator, "{s}/{s}", .{ dir_path, auto_update_stamp_filename }) catch return;
388:    defer allocator.free(stamp_path);
389:
390:    const last_ms = readAutoUpdateStamp(io, stamp_path);
391:    const now_ms = cio.milliTimestamp();
392:    if (!shouldRunAutoUpdate(now_ms, last_ms, env_disabled)) return;
393:
394:    const executable_path = std.process.executablePathAlloc(io, std.heap.page_allocator) catch return;
395:    const thread = std.Thread.spawn(.{}, autoUpdateWorker, .{executable_path}) catch {
396:        std.heap.page_allocator.free(executable_path);
397:        return;
398:    };
399:    thread.detach();
400:
401:    // Only throttle after the worker launches so a local spawn failure can be
402:    // retried on the next MCP startup.
403:    writeAutoUpdateStamp(io, stamp_path, now_ms);
404:}
