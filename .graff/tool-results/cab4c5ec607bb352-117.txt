   307	fn summarizeCorpus(explorer: *Explorer) struct { files: usize, bytes: u64 } {
   308	    explorer.mu.lockShared();
   309	    defer explorer.mu.unlockShared();
   310	
   311	    var files: usize = 0;
   312	    var bytes: u64 = 0;
   313	    var iter = explorer.outlines.iterator();
   314	    while (iter.next()) |entry| {
   315	        files += 1;
   316	        bytes +|= entry.value_ptr.byte_size;
   317	    }
   318	    return .{ .files = files, .bytes = bytes };
