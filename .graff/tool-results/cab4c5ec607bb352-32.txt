    88	fn zigExe() []const u8 {
    89	    // Machines with several zig installs (zigup + homebrew) need the same
    90	    // binary that built the tests; a bare "zig" can resolve to an
    91	    // incompatible version and fail the build these tests spawn.
    92	    return cio.posixGetenv("ZIG") orelse "zig";
    93	}

    99	const EnvVarGuard = struct {
   100	    name: []const u8,
   101	    had_prev: bool,
   102	    prev_len: usize,
   103	    prev: [4096]u8,
   104	
   105	    fn save(name: []const u8) EnvVarGuard {
   106	        var g = EnvVarGuard{ .name = name, .had_prev = false, .prev_len = 0, .prev = undefined };
   107	        if (cio.posixGetenv(name)) |v| {
   108	            if (v.len <= g.prev.len) {
   109	                @memcpy(g.prev[0..v.len], v);
   110	                g.prev_len = v.len;
   111	                g.had_prev = true;
   112	            }
   113	        }
   114	        return g;
   115	    }
   116	
   117	    fn restore(self: *const EnvVarGuard) void {
   118	        if (self.had_prev) {
   119	            cio.posixSetenv(self.name, self.prev[0..self.prev_len]);
   120	        } else {
   121	            cio.posixUnsetenv(self.name);
   122	        }
   123	    }
   124	};

   173	    // 0.17.0-dev.813 rejects it); prefer the env var, which every version
   174	    // honors, and keep the arg off entirely.

   176	        .allocator = testing.allocator,

   179	    });
   180	    defer testing.allocator.free(build.stdout);

   182	
   183	    try testing.expect(build.term == .Exited);
   184	    try testing.expect(build.term.Exited == 0);
   185	}
