   44 | /// Resolve which path to scan from (first indexable root, or cwd fallback) and
   45 | /// fire the deferred scan exactly once. Returns true when this call actually
   46 | /// fired the trigger; false if it had already fired or no usable path is
   47 | /// available. Callers must filter denied paths out of `indexable_roots` first;
   48 | /// the `fallback_cwd` path is policy-checked here.
   49 | pub fn triggerDeferredScanWithFallback(
   50 |     ds: *DeferredScan,
   51 |     indexable_roots: []const Root,
   52 |     fallback_cwd: []const u8,
   53 | ) bool {
   54 |     var path: []const u8 = "";
   55 |     if (indexable_roots.len > 0) {
   56 |         const uri_raw = indexable_roots[0].uri;
   57 |         path = if (std.mem.startsWith(u8, uri_raw, "file://")) uri_raw[7..] else uri_raw;
   58 |     }
   59 |     if (path.len == 0 and fallback_cwd.len > 0 and root_policy.isIndexableRoot(fallback_cwd)) {
   60 |         path = fallback_cwd;
   61 |     }
   62 |     if (path.len == 0) return false;
   63 |     if (ds.triggered.swap(true, .acq_rel)) return false;
   64 |     ds.triggerFn(ds, path);
   65 |     return true;

   85 |         self.mu.lock();

  102 |     /// Takes ownership of `fresh` if it becomes the cache entry. If another
  103 |     /// caller filled the same seq first, frees `fresh` and appends the winner.

  126 |                 if (out.items.len == 0) {
  127 |                     out.items = bytes;

  133 |             }
  134 |             alloc.free(bytes);
  135 |         }
