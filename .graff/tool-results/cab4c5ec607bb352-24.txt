  3241	// ── #619: cli-daemon socket handover ────────────────────────────────────────
  3242	// The reclaim test above covers a serve/mcp daemon retrying until a dead or
  3243	// idle-exited owner lets go. This test covers the other half: an owner that
  3244	// is still LIVE must be actively signalled to give the socket up right away
  3245	// (freshest-wins), instead of the newcomer waiting out a full retry/idle
  3246	// cadence. cliServeConn is a private helper, so the owner side is simulated
  3247	// here with a tiny raw accept loop that mirrors exactly what it does on
  3248	// receipt of cli_yield_sentinel: read the frame, verify it, then give the
  3249	// socket up (close the listener + unlink the path).
  3250	const CliHandoverOwnerSim = struct {
  3251	    fd: c_int,
  3252	    sock_path: []const u8,
  3253	    sd: *std.atomic.Value(bool),
  3254	    saw_sentinel: bool = false,
  3255	    done: std.atomic.Value(bool) = std.atomic.Value(bool).init(false),
  3256	
  3257	    fn run(self: *@This()) void {
  3258	        // Mirror the real accept loop: keep accepting connections and only
  3259	        // give the socket up once one of them is the actual yield sentinel.
  3260	        // A plain liveness probe (cliSocketLive — connect then close with no
  3261	        // data) must NOT be mistaken for a yield request.
  3262	        while (true) {
  3263	            const conn = std.c.accept(self.fd, null, null);
  3264	            if (conn < 0) break;
  3265	            var matched = false;
  3266	            var hdr: [5]u8 = undefined;
  3267	            var off: usize = 0;
  3268	            while (off < hdr.len) {
  3269	                const n = std.c.read(conn, hdr[off..].ptr, hdr.len - off);
  3270	                if (n <= 0) break;
  3271	                off += @intCast(n);
  3272	            }
  3273	            if (off == hdr.len) {
  3274	                const blob_len = std.mem.readInt(u32, hdr[1..5], .little);
  3275	                if (blob_len > 0 and blob_len <= 256) {
  3276	                    var buf: [256]u8 = undefined;
  3277	                    var boff: usize = 0;
  3278	                    while (boff < blob_len) {
  3279	                        const n = std.c.read(conn, buf[boff..].ptr, blob_len - boff);
  3280	                        if (n <= 0) break;
  3281	                        boff += @intCast(n);
  3282	                    }
  3283	                    if (boff == blob_len) {
  3284	                        matched = std.mem.eql(u8, buf[0..blob_len], cli_proxy_mod.cli_yield_sentinel);
  3285	                    }
  3286	                }
  3287	            }
  3288	            _ = std.c.close(conn);
  3289	            if (matched) {
  3290	                self.saw_sentinel = true;
  3291	                break;
  3292	            }
  3293	        }
  3294	        // Mirror the real accept loop's response to a yield request: give the
  3295	        // socket up entirely rather than waiting for an idle timeout.
  3296	        _ = std.c.close(self.fd);
  3297	        var zbuf: [128]u8 = undefined;
  3298	        if (cio.bufPrintZ(&zbuf, "{s}", .{self.sock_path})) |z| {
  3299	            _ = std.c.unlink(z.ptr);
  3300	        } else |_| {}
  3301	        self.sd.store(true, .release);
  3302	        self.done.store(true, .release);
  3303	    }
  3304	};
