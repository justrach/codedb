379:pub fn maybeAutoUpdate(io: std.Io, allocator: std.mem.Allocator) void {
380:    const env_disabled = cio.posixGetenv("CODEDB_NO_AUTO_UPDATE") != null;
381:    const home = cio.homeDir() orelse return;
382:    if (home.len == 0) return;
383:
384:    const dir_path = std.fmt.allocPrint(allocator, "{s}/.codedb", .{home}) catch return;
385:    defer allocator.free(dir_path);
386:    std.Io.Dir.cwd().createDirPath(io, dir_path) catch {};
387:
388:    const stamp_path = std.fmt.allocPrint(allocator, "{s}/{s}", .{ dir_path, auto_update_stamp_filename }) catch return;
389:    defer allocator.free(stamp_path);
390:
391:    const last_ms = readAutoUpdateStamp(io, stamp_path);
392:    const now_ms = cio.milliTimestamp();
393:    if (!shouldRunAutoUpdate(now_ms, last_ms, env_disabled)) return;
394:
395:    const executable_path = std.process.executablePathAlloc(io, std.heap.page_allocator) catch return;
396:    const thread = std.Thread.spawn(.{}, autoUpdateWorker, .{executable_path}) catch {
397:        std.heap.page_allocator.free(executable_path);
398:        return;
399:    };
400:    thread.detach();
401:
402:    // Only throttle after the worker launches so a local spawn failure can be
403:    // retried on the next MCP startup.
404:    writeAutoUpdateStamp(io, stamp_path, now_ms);
405:}
