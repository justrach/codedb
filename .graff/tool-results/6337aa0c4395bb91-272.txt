   76 | /// Detach a daemonized child from its controlling terminal: start a new
   77 | /// session (so it survives the spawning shell / parent CLI) and point
   78 | /// stdin/stdout/stderr at /dev/null (so it never holds the terminal open or
   79 | /// writes stray bytes to it). Best-effort — every step ignores errors, since a
   80 | /// failure here only means the daemon keeps an inherited fd, not that it
   81 | /// malfunctions. Called once at cli-daemon startup. On Windows the equivalent
   82 | /// detachment is requested at spawn time (DETACHED_PROCESS), so this is a no-op.
   83 | pub fn detachFromTerminal() void {
   84 |     if (is_windows or is_freestanding) {
   85 |         return;

  260 | /// Wall-clock nanoseconds since the Unix epoch.
  261 | pub fn nanoTimestamp() i128 {
  262 |     if (is_freestanding) return 0;
  263 |     if (is_windows) {
  264 |         var ft: winsync.FILETIME = undefined;
  265 |         winsync.GetSystemTimePreciseAsFileTime(&ft);
  266 |         // 100-ns ticks since 1601-01-01; shift to the 1970 epoch.
  267 |         const ticks: u64 = (@as(u64, ft.dwHighDateTime) << 32) | ft.dwLowDateTime;
  268 |         return (@as(i128, ticks) - 116444736000000000) * 100;

  270 |         var ts: std.c.timespec = undefined;
  271 |         _ = posix_libc.clock_gettime(CLOCK_REALTIME, &ts);
  272 |         return @as(i128, ts.sec) * 1_000_000_000 + ts.nsec;
  273 |     }
  274 | }

  325 | /// Non-cryptographic random u64 mixing nanotime, PID, and thread ID.
  326 | /// Replaces `std.crypto.random.int(u64)` (removed in 0.16) for tmp-file
  327 | /// suffix collision avoidance. Thread-safe: each thread gets a unique
  328 | /// mix per-call even at the same nanosecond.
  329 | pub fn randU64() u64 {
  330 |     const now: u128 = @bitCast(nanoTimestamp());
  331 |     const ns: u64 = @truncate(now);
  332 |     const sec: u64 = @truncate(now / 1_000_000_000);
  333 |     const tid: u64 = if (is_freestanding) 0 else @intCast(std.Thread.getCurrentId());
  334 |     const pid: u64 = if (is_freestanding)
  335 |         0
  336 |     else if (is_windows)
  337 |         @intCast(winsync.GetCurrentProcessId())
  338 |     else
  339 |         @intCast(std.c.getpid());

  341 |     var x = ns ^ (sec *% 2) ^ (tid *% (1 << 17)) ^ (pid *% (1 << 23));

  343 |     x *%= 0xff51afd7ed558ccd;
  344 |     x ^= x >> 33;

  929 | /// Fire-and-forget spawn: posix_spawnp `argv` with stdin/stdout/stderr

  933 | /// init, so we never reap it (no zombie outlives this short-lived CLI). All
  934 | /// failures are swallowed — auto-spawn is best-effort and the cold path still
  935 | /// produces correct output regardless.

  982 | // ── Memory mapping ─────────────────────────────────────────────────────────
  983 | // Zero-copy read-only file views. POSIX uses mmap(MAP_SHARED, PROT_READ);
  984 | // Windows uses a file-mapping object + MapViewOfFile. Both return a page-aligned
  985 | // const slice that must be released with munmap(). `handle` is the platform file
  986 | // handle — pass `file.handle` from an opened std.Io.File (fd_t on POSIX,
  987 | // windows.HANDLE on Windows; `anytype` keeps the per-target type exact).
  988 | 
  989 | pub const MmapError = error{MapFailed};
  990 | 
  991 | const winmap = if (is_windows) struct {
  992 |     const HANDLE = std.os.windows.HANDLE;
  993 |     const DWORD = std.os.windows.DWORD;
  994 |     const BOOL = std.os.windows.BOOL;
  995 |     const PAGE_READONLY: DWORD = 0x02;
  996 |     const FILE_MAP_READ: DWORD = 0x0004;
  997 |     extern "kernel32" fn CreateFileMappingW(hFile: HANDLE, lpAttributes: ?*anyopaque, flProtect: DWORD, dwMaximumSizeHigh: DWORD, dwMaximumSizeLow: DWORD, lpName: ?[*:0]const u16) callconv(.winapi) ?HANDLE;
  998 |     extern "kernel32" fn MapViewOfFile(hFileMappingObject: HANDLE, dwDesiredAccess: DWORD, dwFileOffsetHigh: DWORD, dwFileOffsetLow: DWORD, dwNumberOfBytesToMap: usize) callconv(.winapi) ?*anyopaque;
  999 |     extern "kernel32" fn UnmapViewOfFile(lpBaseAddress: *const anyopaque) callconv(.winapi) BOOL;
 1000 |     extern "kernel32" fn CloseHandle(hObject: HANDLE) callconv(.winapi) BOOL;
 1001 | } else struct {};
 1002 | 
 1003 | /// mmap `len` bytes of `handle` read-only. Returns a page-aligned const view.
 1004 | pub fn mmapReadonly(handle: anytype, len: usize) MmapError![]align(std.heap.page_size_min) const u8 {
 1005 |     if (is_windows) {
 1006 |         // A 0 max-size maps the whole file; the view stays valid after the
 1007 |         // mapping handle is closed, so we close it eagerly.
 1008 |         const mapping = winmap.CreateFileMappingW(handle, null, winmap.PAGE_READONLY, 0, 0, null) orelse
 1009 |             return error.MapFailed;
 1010 |         defer _ = winmap.CloseHandle(mapping);
 1011 |         const base = winmap.MapViewOfFile(mapping, winmap.FILE_MAP_READ, 0, 0, len) orelse
 1012 |             return error.MapFailed;
 1013 |         const ptr: [*]align(std.heap.page_size_min) const u8 = @ptrCast(@alignCast(base));
 1014 |         return ptr[0..len];
 1015 |     } else {
 1016 |         return std.posix.mmap(null, len, .{ .READ = true }, .{ .TYPE = .SHARED }, handle, 0) catch
 1017 |             return error.MapFailed;
 1018 |     }
 1019 | }

 1024 |     if (is_windows) {
 1025 |         _ = winmap.UnmapViewOfFile(data.ptr);
 1026 |     } else {
 1027 |         std.posix.munmap(data);
 1028 |     }
