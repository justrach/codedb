  6620	                }

  6624	        std.mem.sort(Entry, top.items, {}, struct {

  6630	        // We dropped the explorer lock above. The path refs are stable across

  6678	    fn parsePythonLine(self: *Explorer, line: []const u8, line_num: u32, outline: *FileOutline) !void {
  6679	        const a = self.allocator;
  6680	        if (startsWith(line, "def ")) {
  6681	            if (extractIdent(line[4..])) |name| {
  6682	                try appendOutlineSymbol(a, outline, name, .function, line_num, line);
  6683	            }
  6684	        } else if (startsWith(line, "class ")) {
  6685	            if (extractIdent(line[6..])) |name| {
  6686	                try appendOutlineSymbol(a, outline, name, .class_def, line_num, line);
  6687	            }
  6688	        } else if (startsWith(line, "import ") or startsWith(line, "from ")) {
  6689	            try appendOutlineSymbol(a, outline, line, .import, line_num, null);
  6690	            if (startsWith(line, "from ")) {
  6691	                // "from mypackage.utils import X" -> mypackage/utils.py (single module dep)
  6692	                if (extractPythonModulePath(line)) |mod_path| {
  6693	                    try appendPythonModuleDep(a, outline, mod_path);
  6694	                }
  6695	            } else {
  6696	                // "import a, b.c, d as e" -- record one dep per comma-separated module.
  6697	                const rest = std.mem.trimStart(u8, line[7..], " \t");
  6698	                var it = std.mem.splitScalar(u8, rest, ',');
  6699	                while (it.next()) |raw| {
  6700	                    var mod = std.mem.trim(u8, raw, " \t");
  6701	                    if (std.mem.indexOf(u8, mod, " as ")) |as_pos| mod = std.mem.trimEnd(u8, mod[0..as_pos], " \t");
  6702	                    if (std.mem.indexOfScalar(u8, mod, ' ')) |sp| mod = mod[0..sp];
  6703	                    if (mod.len == 0 or mod[0] == '.') continue;
  6704	                    try appendPythonModuleDep(a, outline, mod);
  6705	                }
  6706	            }
  6707	        }
  6708	    }
  6709	    fn parseTsLine(self: *Explorer, line: []const u8, line_num: u32, outline: *FileOutline) !void {
  6710	        const a = self.allocator;
  6711	        if (containsAny(line, &.{ "function ", "const ", "let ", "var ", "class ", "interface ", "enum ", "type " })) {
  6712	            const kind: SymbolKind = if (std.mem.indexOf(u8, line, "function") != null)
  6713	                .function
  6714	            else if (std.mem.indexOf(u8, line, "class ") != null)
  6715	                .class_def
  6716	            else if (std.mem.indexOf(u8, line, "interface ") != null)
  6717	                .interface_def
  6718	            else if (std.mem.indexOf(u8, line, "enum ") != null)
  6719	                .enum_def
  6720	            else if (std.mem.indexOf(u8, line, "type ") != null)
  6721	                .type_alias
  6722	            else
  6723	                .constant;
  6724	            const trimmed = skipKeywords(line);
  6725	            if (extractIdent(trimmed)) |name| {
  6726	                try appendOutlineSymbol(a, outline, name, kind, line_num, line);
  6727	            }
  6728	        }
  6729	        if (startsWith(line, "import ") or containsAny(line, &.{"require("})) {
  6730	            try appendOutlineSymbol(a, outline, line, .import, line_num, null);
  6731	            if (extractStringLiteral(line)) |path| {
  6732	                try appendImportPath(a, outline, path);
  6733	            }
  6734	        } else if (startsWith(line, "} from ") or
  6735	            (startsWith(line, "export ") and containsAny(line, &.{ " from \"", " from '" })))
  6736	        {
  6737	            // Closing line of a multi-line import/export, e.g.
  6738	            //   import {
  6739	            //     a, b,
  6740	            //   } from "../mod.ts";
  6741	            // or a re-export like `export * from "./x.ts"`. The `from "..."`
  6742	            // clause lands on a line that does not contain "import ", so without
  6743	            // this branch the path is dropped (only single-line imports were
  6744	            // captured). Gated on these concrete shapes so arbitrary `from "..."`
  6745	            // inside strings/comments (e.g. SQL `select * from "users"`) is not
  6746	            // mistaken for a dependency.
  6747	            if (extractStringLiteral(line)) |path| {
  6748	                try appendImportPath(a, outline, path);
  6749	            }
  6750	        }
  6751	    }
